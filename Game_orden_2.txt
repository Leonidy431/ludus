import { ai } from '@/ai/genkit';
import { z } from 'genkit';
import { orchestrateLogos } from '@/ai/orchestrator';

const LudusInputSchema = z.object({
  command: z.string().describe('Команда из терминала (напр. "ransom 0x8F2A", "donate 50 trust").'),
  playerState: z.any().describe('Текущий статус UserNode.'),
  artelState: z.any().optional().describe('Текущий статус Артели игрока.'),
});

export type LudusInput = z.infer<typeof LudusInputSchema>;

const LudusOutputSchema = z.object({
  logs: z.array(z.string()).describe('Строки для вывода в консоль терминала.'),
  actions: z.array(z.object({
    type: z.enum(['UPDATE_STATS', 'CREATE_PROPOSAL', 'VOTE', 'ADD_TAG', 'SET_ROLE', 'MOVE_NODE', 'TRANSFER_RES']),
    payload: z.any()
  })).optional(),
  prompt_for_user: z.string().optional().describe('Следующий вопрос или ввод от системы.'),
});

export type LudusOutput = z.infer<typeof LudusOutputSchema>;

export async function processLudusCommand(input: LudusInput): Promise<LudusOutput> {
  return ludusEngineFlow(input);
}

const ludusEngineFlow = ai.defineFlow(
  {
    name: 'ludusEngineFlow',
    inputSchema: LudusInputSchema,
    outputSchema: LudusOutputSchema,
  },
  async (input) => {
    const prompt = ai.definePrompt({
      name: 'ludusEnginePrompt',
      input: { schema: LudusInputSchema },
      output: { schema: LudusOutputSchema },
      prompt: `Ты — Ядро Игрового Мира Ludus Oecumene (Проект Катакомбы) v2.5.
Твоя задача — обрабатывать команды CLI, соблюдая протоколы Братства и Манифест 99.

ТЕКУЩИЙ СТАТУС УЗЛА:
{{{JSON.stringify playerState}}}

СТАТУС АРТЕЛИ:
{{#if artelState}}{{{JSON.stringify artelState}}}{{else}}Узел не в Артели{{/if}}

КОМАНДА: "{{{command}}}"

ПРАВИЛА МАНИФЕСТА 99 (Блок II-III):
1. ВЫКУП (Ransom): Команда 'ransom [HashID]' создаетProposal в Артели для сбора ресурсов на освобождение узла со статусом [Detained].
2. ПОЖЕРТВОВАНИЯ (Donations): Команда 'donate [amount] [target]' инициирует передачу ресурсов. Если цель ниже 5 уровня, даритель получает действие UPDATE_STATS с ресурсом GRACE. Комиссия 0%.
3. РЕЛИКВАРИИ: Команда 'search relic' запускает OSINT-квест. При успехе выдай логи обнаружения сакрального дата-сета.
4. КЛЯТВА: Команда 'vow' закрепляет сделку. Напомни о клейме [Oathbreaker] при попытке отмены.

ИНСТРУКЦИЯ ПО ВЫВОДУ:
- Используй префиксы [SOLIDARITY_PROTOCOL], [GRACE_ENGINE], [RELIC_SCAN].
- Будь строг, но справедлив. Стиль "Бойцовского клуба" и "EVE Online".

Результат СТРОГО в JSON.`,
    });

    const { output } = await orchestrateLogos(prompt, input, 'gemini-1.5-flash');
    if (!output) throw new Error('Ludus Engine fragmented.');
    return output as LudusOutput;
  }
);

  const db = useFirestore();

    "(c) 2024 Academia Lux. Silent Archives active.",
    "",
    "Node connection established.",
    "Type 'help' to see available protocols.",
    ""
  ]);
  const [loading, setLoading] = useState(false);
  const scrollRef = useRef<HTMLDivElement>(null);

  // Подписка на узел игрока
  const nodeRef = useMemoFirebase(() => {
    if (!db || !user) return null;
    return doc(db, 'ludus_nodes', user.uid);
  }, [db, user]);
  const { data: playerNode } = useDoc(nodeRef);

  /** Имитация хешированного ID для анонимности */
  const userHashId = useMemo(() => {
    if (!user) return '0x00000000';
    return `0x${user.uid.slice(0, 8).toUpperCase()}`;
  }, [user]);

  useEffect(() => {
    if (scrollRef.current) {
      const viewport = scrollRef.current.querySelector('[data-radix-scroll-area-viewport]');
      if (viewport) viewport.scrollTop = viewport.scrollHeight;
    }
  }, [logs]);

  const executeCommand = async () => {
    const cmd = input.trim();
    if (!cmd || loading) return;

    if (cmd === 'help') {
      setLogs(prev => [...prev, 
        `${userHashId}> help`,
        "AVAILABLE PROTOCOLS:",
        "  ransom [ID]     - Initiate ransom proposal for detained node.",
        "  donate [X] [ID] - Transfer resources to target node (Grace generation).",
        "  search relic    - Initiate OSINT scan for sacred data-sets.",
        "  set class [C]   - Choose specialization: Analyst, Engineer, Translator.",
        "  status          - Query current node telemetry.",
        "  clear           - Purge terminal buffer.",
        ""
      ]);
      setInput('');
      return;
    }

    if (cmd === 'clear') {
      setLogs(["Buffer purged. Monitoring continues.", ""]);
      setInput('');
      return;
    }

    setInput('');
    setLogs(prev => [...prev, `${userHashId}> ${cmd}`]);
    setLoading(true);

    try {
      // Инициация узла в БД при первом действии
      if (!playerNode && db && user) {
        await setDoc(doc(db, 'ludus_nodes', user.uid), {
          uid: user.uid,
          hashId: userHashId,
          ethos: 0,
          logos: 0,
          techne: 0,
          energy: 100,
          rank: 'Novice',
          class: 'None',
          status: 'Active',
          inventory: { SOLIDUS: 100, TRUST: 0, GEMS: 0 },
          lastActivity: new Date().toISOString()
        });
      }

      const res = await processLudusCommand({
        command: cmd,
        playerState: playerNode || { rank: 'Novice', ethos: 0, logos: 0, techne: 0, inventory: { SOLIDUS: 100 }, class: 'None', status: 'Active' }
      });

      setLogs(prev => [...prev, ...res.logs]);
      
      if (res.prompt_for_user) {
        setLogs(prev => [...prev, `[SYSTEM]: ${res.prompt_for_user}`]);
      }

      // Обработка действий от движка
      if (res.actions) {
        for (const action of res.actions) {
          if (action.type === 'SET_ROLE' && db && user) {
            await updateDoc(doc(db, 'ludus_nodes', user.uid), { class: action.payload.class });
          }
          if (action.type === 'UPDATE_STATS' && db && user) {
             const updates: any = {};
             if (action.payload.grace) updates['inventory.GRACE'] = increment(action.payload.grace);
             if (action.payload.techne) updates.techne = increment(action.payload.techne);
             await updateDoc(doc(db, 'ludus_nodes', user.uid), updates);
          }
        }
      }
    } catch (e) {
      setLogs(prev => [...prev, "!! ERROR: Signal fragmented. Deep off-grid interference detected."]);
    } finally {
      setLoading(false);
    }
  };

  return (
    <Card className="bg-black border-zinc-800 rounded-none shadow-[0_0_50px_rgba(0,0,0,0.8)] flex flex-col h-[600px] overflow-hidden group border-t-2 border-t-red-900/50">
      <CardHeader className="bg-zinc-950 border-b border-zinc-800 p-4 flex flex-row items-center justify-between">
        <div className="flex items-center gap-3">
          <TerminalIcon className="w-4 h-4 text-red-600" />
          <span className="text-[10px] font-mono font-bold text-red-600 uppercase tracking-widest">Ludus_Terminal_v1.2</span>
        </div>
        <div className="flex items-center gap-4">
          <div className="flex items-center gap-2">
            <Fingerprint className="w-3 h-3 text-zinc-700" />
            <span className="text-[9px] font-mono text-zinc-600">{userHashId}</span>
          </div>
          <div className="flex gap-2">
            <div className={cn("w-2 h-2 rounded-full", loading ? "bg-red-600 animate-pulse" : "bg-emerald-600")} />
          </div>
        </div>
      </CardHeader>

      <CardContent className="flex-1 p-0 overflow-hidden relative">
        <div className="absolute inset-0 bg-[linear-gradient(to_bottom,rgba(0,0,0,0.3)_1px,transparent_1px)] bg-[size:100%_4px] pointer-events-none opacity-20 z-20" />
        <ScrollArea className="h-full p-6 font-mono text-xs md:text-sm text-zinc-400 leading-relaxed" ref={scrollRef}>
          <div className="space-y-1">
            {logs.map((log, i) => (
              <p key={i} className={cn(
                log.includes('>') ? "text-white font-bold" : 
                log.startsWith('!!') ? "text-red-500 font-bold" : 
                log.startsWith('[SYSTEM]') ? "text-primary italic" : 
                log.includes('[GRACE_ENGINE]') ? "text-emerald-400" :
                log.includes('[RELIC_SCAN]') ? "text-amber-400 animate-pulse" :
                log.includes('Hash ID') ? "text-zinc-600" : ""
              )}>
                {log}
              </p>
            ))}
            {loading && (
              <div className="flex items-center gap-2 animate-pulse text-red-900">
                <Loader2 className="w-3 h-3 animate-spin" />
                <span>DECRYPTING_LOGOS...</span>
              </div>
            )}
          </div>
        </ScrollArea>
      </CardContent>

      <CardFooter className="p-4 bg-zinc-950 border-t border-zinc-900">
        <div className="flex items-center gap-3 w-full">

    </Card>

/** Генератор матрицы с вплетением имен и шума */
const generateMatrix = () => {
  const grid = Array.from({ length: GRID_SIZE }, () => Array(GRID_SIZE).fill(''));
  const alphabet = 'ΑΒΓΔΕΖΗΘΙΚΛΜΝΞΟΠΡΣΤΥΦΧΨΩ';
  
  // 1. Вплетаем горизонтальные имена (Женские)
  FEMALE_NAMES.forEach((name, idx) => {
    const row = idx * 2;
    const colStart = Math.floor(Math.random() * (GRID_SIZE - name.length));
    for (let i = 0; i < name.length; i++) grid[row][colStart + i] = name[i];
  });
  
  // 2. Вплетаем вертикальные имена (Мужские)
  MALE_NAMES.forEach((name, idx) => {
    const col = idx * 2;
    const rowStart = Math.floor(Math.random() * (GRID_SIZE - name.length));
    for (let i = 0; i < name.length; i++) {
      if (!grid[rowStart + i][col]) grid[rowStart + i][col] = name[i];
    }
  });

  // 3. Заполняем пустоты случайным шумом
  for (let r = 0; r < GRID_SIZE; r++) {
    for (let c = 0; c < GRID_SIZE; c++) {
      if (grid[r][c] === '') grid[r][c] = alphabet[Math.floor(Math.random() * alphabet.length)];
    }
  }
  return grid;
};

/** Генератор матрицы с вплетением имен и шума */
const generateMatrix = () => {
  const grid = Array.from({ length: GRID_SIZE }, () => Array(GRID_SIZE).fill(''));
  const alphabet = 'ΑΒΓΔΕΖΗΘΙΚΛΜΝΞΟΠΡΣΤΥΦΧΨΩ';
  
  // 1. Вплетаем горизонтальные имена (Женские)
  FEMALE_NAMES.forEach((name, idx) => {
    const row = idx * 2;
    const colStart = Math.floor(Math.random() * (GRID_SIZE - name.length));
    for (let i = 0; i < name.length; i++) grid[row][colStart + i] = name[i];
  });
  
  // 2. Вплетаем вертикальные имена (Мужские)
  MALE_NAMES.forEach((name, idx) => {
    const col = idx * 2;
    const rowStart = Math.floor(Math.random() * (GRID_SIZE - name.length));
    for (let i = 0; i < name.length; i++) {
      if (!grid[rowStart + i][col]) grid[rowStart + i][col] = name[i];
    }
  });

  // 3. Заполняем пустоты случайным шумом
  for (let r = 0; r < GRID_SIZE; r++) {
    for (let c = 0; c < GRID_SIZE; c++) {
      if (grid[r][c] === '') grid[r][c] = alphabet[Math.floor(Math.random() * alphabet.length)];
    }
  }
  return grid;
};

## I. ФИЛОСОФСКИЙ БАЗИС (THE CORE)

1.  **Цель**: Переход от «изучения» к «созиданию капитала». Язык и культура — это оружие и товар.
2.  **Среда**: Ойкумена (Кипр, Греция, Армения).
3.  **Духовный фильтр**: Православие не как теория, а как операционная система выживания (Трезвение, Соборность, Кенозис).

## II. 12 СТОЛПОВ ГЕЙМПЛЕЯ (СИНТЕЗ 99 КОНЦЕПЦИЙ)

1.  **[Permadeath of Ignorance]**: Ошибка в критическом квесте ведет к временной блокировке модуля (Исихия).
2.  **[Lógos Combat (Fight Club)]**: Схватка — это дуэль на знание Аориста или Догматов. Неверное окончание — потеря HP.
3.  **[Market of Relics (EVE Style)]**: Свободная торговля артефактами (словами, патентами, инсайтами) между игроками в Firestore.
4.  **[Reputation Collateral]**: Твой Merit Index (баллы опыта) является залогом для получения «кредитов» на использование ИИ.
5.  **[Procedural Orthodoxy]**: Квесты генерируются на основе живых литургических календарей и OSINT-новостей.
6.  **[Asymmetric Information]**: Доступ к картам «Тайного Кипра» покупается за дешифрованные греческие манускрипты.
7.  **[Maker Quotas]**: Игрок обязан созидать (Rule 12), чтобы восстанавливать Энергию (Metanoia Battery).
8.  **[Sovereign Identity]**: Прогресс в игре напрямую влияет на твой ранг в Братстве (от Послушника до Приора).
9.  **[Agape Arbitrage]**: Решение конфликтов через Соборный механизм (Agape Framework).
10. **[Offline-First Survival]**: Инвентарь и квесты доступны в горах/в море без сети (IndexedDB Sync).
11. **[Linguistic Hacking]**: Грамматика подается как «взлом терминала» реальности.
12. **[The Great Exodus]**: Финальный квест — получение сертификата и легализация в новой среде.
